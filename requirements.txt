# ─────────────────────────────────────────────────────────────────────────────
# fpvlink/requirements.txt
#
# Python runtime dependencies for the FPVLink capture layer and pipeline driver.
#
# Install with:
#   pip3 install -r requirements.txt
#
# On the Orange Pi 5 Plus (Debian/Armbian):
#   sudo apt install python3-pip python3-gi python3-gi-cairo gir1.2-gst-plugins-base-1.0
#   pip3 install --user -r requirements.txt
#
# This list is deliberately short. Everything else the capture layer uses —
# gi (GStreamer bindings), asyncio, threading, struct, ctypes — is either a
# system package or the Python standard library, so it must not be listed here.
# ─────────────────────────────────────────────────────────────────────────────

# ─── USB I/O ──────────────────────────────────────────────────────────────────

# pyusb: USB access via libusb-1.0. Used by the capture layer to open the DJI
# FPV Goggles bulk endpoints, run the handshake, and read the raw bytestream.
#
# System prerequisite: libusb-1.0-0 (usually pre-installed)
#   sudo apt install libusb-1.0-0
#
# udev rule (allows the fpvlink user to access USB without root):
#   SUBSYSTEM=="usb", ATTRS{idVendor}=="2ca3", MODE="0660", GROUP="plugdev"
#
# Docs: https://pyusb.github.io/pyusb/
pyusb>=1.2.1

# ─────────────────────────────────────────────────────────────────────────────
# System-level packages (install via apt, NOT pip)
# ─────────────────────────────────────────────────────────────────────────────
#
# GStreamer Python bindings (gi / PyGObject):
#   sudo apt install python3-gi python3-gi-cairo gir1.2-gst-plugins-base-1.0
#
# GStreamer core + plugins:
#   sudo apt install \
#     gstreamer1.0-tools \
#     gstreamer1.0-plugins-base \
#     gstreamer1.0-plugins-good \
#     gstreamer1.0-plugins-bad \
#     gstreamer1.0-plugins-ugly \
#     gstreamer1.0-libav \
#     gstreamer1.0-rkmpp
#
# DUML framing, binary parsing, config loading, and logging are all implemented
# directly against the standard library (struct / json / logging). Earlier
# revisions of this file listed construct, aiohttp, python-dotenv and structlog;
# none of them were ever imported, so pip no longer installs them on the device.
